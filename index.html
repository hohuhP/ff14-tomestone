async function fetchData(saved) { try { // 서버 한글 이름을 영문 슬러그로 변환하는 매핑 테이블 const serverMap = { "카벙클": "Carbuncle", "초코보": "Chocobo", "모그리": "Moogle", "톤베리": "Tonberry", "펜리르": "Fenrir" }; // 입력된 값이 한글이면 영문으로 바꾸고, 아니면 그대로 사용 const serverSlug = serverMap[saved.server] || saved.server; const tokenRes = await fetch('https://www.fflogs.com/oauth/token', { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: `grant_type=client_credentials&client_id=${saved.id}&client_secret=${saved.secret}` }); const tokenData = await tokenRes.json(); const token = tokenData.access_token; if(!token) throw new Error("API 인증 실패. ID/Secret을 확인하세요."); const query = `query { characterData { character(name: "${saved.name}", serverSlug: "${serverSlug}", serverRegion: "KR") { recentReports(limit: 100) { data { code title masterData { actors(type: "Player") { id name } } fights { id kill name friendlyPlayers } } } } } }`; const res = await fetch('https://www.fflogs.com/api/v2/client', { method: 'POST', headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ query }) }); const result = await res.json(); if (!result.data || !result.data.characterData || !result.data.characterData.character) { throw new Error(`'${saved.name}' 캐릭터를 ${serverSlug} 서버에서 찾을 수 없습니다.`); } const reports = result.data.characterData.character.recentReports.data || []; const processedFights = []; reports.forEach(report => { if (!report.fights) return; report.fights.forEach(fight => { if (fight.friendlyPlayers && fight.friendlyPlayers.length === 8) { const partyNames = fight.friendlyPlayers.map(pId => { const actor = report.masterData.actors.find(a => a.id === pId); return actor ? actor.name : "Unknown"; }); processedFights.push({ reportCode: report.code, title: report.title, fight, party: partyNames }); } }); }); renderFights(processedFights, saved.name); } catch (e) { // ... 에러 출력 로직 (이전과 동일) } }
