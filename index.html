async function fetchData(saved) { try { const tokenRes = await fetch('https://www.fflogs.com/oauth/token', { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: `grant_type=client_credentials&client_id=${saved.id}&client_secret=${saved.secret}` }); const tokenData = await tokenRes.json(); const token = tokenData.access_token; // 쿼리 수정: 각 전투(fight)의 참여자(fights -> participants) 정보를 직접 가져옵니다. const query = `query { characterData { character(name: "${saved.name}", serverSlug: "${saved.server}", serverRegion: "KR") { recentReports(limit: 100) { data { code title masterData { actors(type: "Player") { id name } } fights { id kill name friendlyPlayers /* 이 전투에 참여한 플레이어 ID 리스트 */ } } } } } }`; const res = await fetch('https://www.fflogs.com/api/v2/client', { method: 'POST', headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ query }) }); const result = await res.json(); const allReports = result.data.characterData.character.recentReports.data; const processedFights = []; allReports.forEach(report => { report.fights.forEach(fight => { // 핵심: 리포트 전체 인원이 아니라, 해당 '전투'에 참여한 인원이 8명인지 확인 if (fight.friendlyPlayers && fight.friendlyPlayers.length === 8) { // 참여자 ID를 이름으로 변환 const partyNames = fight.friendlyPlayers.map(playerId => { const actor = report.masterData.actors.find(a => a.id === playerId); return actor ? actor.name : "Unknown"; }); processedFights.push({ reportCode: report.code, title: report.title, fight: fight, party: partyNames }); } }); }); renderFights(processedFights, saved.name); } catch (e) { console.error(e); document.getElementById('main-content').innerHTML = `

데이터 로딩 실패

`; } } function renderFights(fights, myName) { const container = document.getElementById('main-content'); container.innerHTML = fights.length === 0 ? "

" : ""; fights.forEach(item => { const { reportCode, title, fight, party } = item; const uniqueId = `${reportCode}_${fight.id}`; const isKill = fight.kill; const savedData = JSON.parse(localStorage.getItem('memo_' + uniqueId)) || {w:'', m:''}; const card = document.createElement('div'); card.className = 'log-card'; card.innerHTML = `
