async function loadCharData() { const name = document.getElementById('cnick').value.trim(); const server = document.getElementById('sv-select').value; const btn = document.getElementById('search-btn'); if(!name) return alert("닉네임을 입력하세요."); btn.innerText = "로딩 중..."; // 수정된 쿼리: 에러가 발생한 rankingType 인자를 제거하고 구조를 단순화함 const q = `query { characterData { character(name: "${name}", serverSlug: "${server}", serverRegion: "KR") { name zoneRankings recentReports(limit: 15) { data { code title fights(killType: Kills) { id name difficulty } } } } } }`; try { const res = await fetch("https://www.fflogs.com/api/v2/client", { method: "POST", headers: { "Content-Type": "application/json", "Authorization": `Bearer ${authToken}` }, body: JSON.stringify({ query: q }) }); const json = await res.json(); if(json.errors) { console.error("GraphQL Error Details:", json.errors); // 만약 zoneRankings에서 또 에러가 날 경우를 대비해 해당 필드 없이 재시도하는 로직을 넣거나 에러를 출력합니다. return alert("API 응답 오류: " + json.errors[0].message); } const char = json.data?.characterData?.character; if(!char) { return alert(`캐릭터를 찾을 수 없습니다.\n이름: ${name}\n서버: ${server}\n\n위 정보가 FF Logs 홈페이지에 등록된 것과 일치하는지 확인하세요.`); } // UI 업데이트: zoneRankings 구조에 따라 안전하게 데이터 추출 const as = char.zoneRankings || {}; // All-Star 데이터가 없을 경우를 대비한 기본값 처리 const rankPercent = as.rankPercent || 0; const rankPoints = as.rankPoints || 0; document.getElementById('top-user-bar').style.display = 'flex'; document.getElementById('as-pct').innerText = rankPercent.toFixed(1); document.getElementById('as-pct').style.background = getC(rankPercent); document.getElementById('as-name').innerText = char.name; document.getElementById('as-server').innerText = server; document.getElementById('as-points').innerText = rankPoints.toLocaleString(); reports = char.recentReports.data; document.getElementById('tab-nav').style.display = 'flex'; document.getElementById('fight-list').style.display = 'grid'; sortFights('all', document.querySelector('.tab')); } catch(e) { console.error("Fetch Error:", e); alert("데이터를 불러오는 중 기술적인 문제가 발생했습니다."); } finally { btn.innerText = "불러오기"; } }
