async function fetchData(saved) { try { const tokenRes = await fetch('https://www.fflogs.com/oauth/token', { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: `grant_type=client_credentials&client_id=${saved.id}&client_secret=${saved.secret}` }); const tokenData = await tokenRes.json(); const token = tokenData.access_token; const query = `query { characterData { character(name: "${saved.name}", serverSlug: "${saved.server}", serverRegion: "KR") { recentReports(limit: 100) { /* 검색 범위를 100개로 확대 */ data { code title fights { kill } masterData { actors(type: "Player") { name } } } } } } }`; const res = await fetch('https://www.fflogs.com/api/v2/client', { method: 'POST', headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ query }) }); const result = await res.json(); const allReports = result.data.characterData.character.recentReports.data; // 필터링: 플레이어 수가 8명인 것 (데이터 구조에 따라 유연하게 체크) const eightManReports = allReports.filter(r => { const playerCount = r.masterData?.actors?.length || 0; return playerCount === 8; }); renderLogs(eightManReports); } catch (e) { console.error(e); document.getElementById('main-content').innerHTML = `

데이터를 불러오는 중 오류가 발생했습니다.

`; } }
