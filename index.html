async function fetchData(saved) { try { const tokenRes = await fetch('https://www.fflogs.com/oauth/token', { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: `grant_type=client_credentials&client_id=${saved.id}&client_secret=${saved.secret}` }); if (!tokenRes.ok) { throw new Error("API 인증 실패: Client ID와 Secret을 다시 확인하세요."); } const tokenData = await tokenRes.json(); const token = tokenData.access_token; const query = `query { characterData { character(name: "${saved.name}", serverSlug: "${saved.server}", serverRegion: "KR") { recentReports(limit: 20) { data { code title masterData { actors(type: "Player") { id name icon server type } } fights { id kill name friendlyPlayers boss difficulty } } } } } }`; const res = await fetch('https://www.fflogs.com/api/v2/client', { method: 'POST', headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ query }) }); const result = await res.json(); if (result.errors) { throw new Error("GraphQL 에러: 캐릭터 이름이나 서버 정보를 확인하세요."); } if (!result.data.characterData.character) { throw new Error("캐릭터를 찾을 수 없습니다. 이름과 서버가 정확한지 확인하세요."); } renderFights(result.data.characterData.character.recentReports.data, saved); } catch (e) { console.error(e); alert(e.message); // 화면에 에러 메시지 출력 localStorage.clear(); // 잘못된 정보면 초기화 location.reload(); } }
